\midrule

\code{PendingRequest} stores \code{to}, \code{peer}, \code{kind} (\code{RequestVote} / \code{AppendEntries} / snapshot), \code{sent\_at}, and a copy of \code{msg}.

\subsection{\code{execute\_actions}}

Loop over each \code{Action}:
\begin{itemize}[nosep]
  \item \code{Send}: if \code{inbound\_from = Some((from, channel\_id))} and \code{to == from} $\rightarrow$ \code{SendResponse} + \code{NotifyHandler::One(conn)}; otherwise \code{send\_rpc}.
  \item \code{Broadcast}: \code{send\_rpc} to each of \code{other\_voters()}.
  \item Role / Apply / Snapshot $\rightarrow$ \code{GenerateEvent}.
\end{itemize}

\subsection{\code{send\_rpc}}

\begin{enumerate}[label=\arabic*., nosep]
  \item \code{peer\_map.peer\_id(to)} --- unknown $\rightarrow$ \code{handle\_rpc\_failure}, \code{RpcFailed}.
  \item Not \code{is\_connected(peer)} $\rightarrow$ failure + \code{dial\_seed} (lossy; no RpcFailed is emitted on the AE retry path).
  \item Allocate a \code{correlation\_id}, \code{pending.insert}, queue \code{NotifyHandler::Any} + \code{SendRequest}.
\end{enumerate}

\subsection{Public events for the application}

\code{RoleChanged}, \code{Committed}, \code{PeerMapped} (startup), \code{SnapshotNeeded}, \code{SnapshotInstalled}, \code{RpcFailed}.

\section{\code{RaftBehaviour::poll} ordering}

\begin{enumerate}[label=\textbf{B\arabic*.}, nosep]
  \item Pop \code{pending\_events} $\rightarrow$ \code{Poll::Ready(ToSwarm)}.
  \item Scan \code{pending} for entries past \code{rpc\_timeout} $\rightarrow$ \code{fail\_pending} $\rightarrow$ engine \code{handle\_rpc\_failure}.
  \item If \code{now >= engine.next\_deadline()} $\rightarrow$ \code{tick} + \code{execute\_actions}.
  \item Poll \code{Sleep}; Ready $\rightarrow$ re-arm \code{earliest\_wake} and loop; Pending $\rightarrow$ return.
\end{enumerate}

The next wake-up deadline:
\begin{lstlisting}
earliest_wake = min(
    engine.next_deadline(),
    min_pending(sent_at + rpc_timeout),
)
\end{lstlisting}

\code{pending\_events} are drained \textbf{first} so that Dial/NotifyHandler/Event are not starved under load.
